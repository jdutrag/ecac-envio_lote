import pip_system_certs.wrapt  # ESSENCIAL: Deve ser a primeira linha para integrar com o Windows
import requests
import os
import glob
import time

# --- CONFIGURAÇÕES DO AMBIENTE ---
# URL para recepção de lotes da EFD-REINF (Ambiente de Produção)
URL_REINF = "https://reinf.receita.economia.gov.br/recepcao/lotes"
# Pasta onde estão os XMLs gerados pelo seu script anterior
PASTA_XMLS = "envios" 

def transmitir_lotes_reinf():
    """
    Função para transmitir arquivos XML utilizando o certificado A3 
    instalado no Windows (AC Defesa) via integração Schannel.
    """
    # Verifica se a pasta de envios existe
    if not os.path.exists(PASTA_XMLS):
        print(f"Erro: A pasta '{PASTA_XMLS}' não foi encontrada.")
        return

    # Busca todos os arquivos .xml na pasta
    arquivos = glob.glob(os.path.join(PASTA_XMLS, "*.xml"))
    
    if not arquivos:
        print("Nenhum arquivo XML encontrado para transmissão.")
        return

    print(f"Iniciando transmissão de {len(arquivos)} arquivos...")

    for arquivo_path in arquivos:
        nome_arquivo = os.path.basename(arquivo_path)
        
        with open(arquivo_path, 'rb') as f:
            conteudo_xml = f.read()

        headers = {
            'Content-Type': 'application/xml',
            'User-Agent': 'Mozilla/5.0' # Simula um navegador para evitar bloqueios simples
        }

        try:
            # O 'requests' agora usará automaticamente o certificado do Windows
            # O sistema deve abrir a janela para você digitar o PIN do Token A3
            response = requests.post(
                URL_REINF, 
                data=conteudo_xml, 
                headers=headers,
                timeout=30
            )

            if response.status_code == 201 or response.status_code == 200:
                print(f"[SUCESSO] {nome_arquivo} enviado com sucesso.")
            else:
                print(f"[ERRO {response.status_code}] Falha ao enviar {nome_arquivo}: {response.text}")

        except requests.exceptions.SSLError as e:
            print(f"[FALHA SSL] Erro de certificado no arquivo {nome_arquivo}. Verifique se o Token A3 está conectado.")
            print(f"Detalhes: {e}")
        except Exception as e:
            print(f"[ERRO CRÍTICO] Falha inesperada ao processar {nome_arquivo}: {e}")
        
        # Pequena pausa para não sobrecarregar a conexão entre lotes
        time.sleep(0.5)

if __name__ == "__main__":
    # Lembre-se de ter instalado: pip install pip-system-certs requests
    transmitir_lotes_reinf()
